# Cerebras (First fallback, free)

# Mistral (Second fallback, free — 1B tokens/month)
# Get key: https://console.mistral.ai
MISTRAL_API_KEY=your-mistral-key-here

# OpenRouter (Third fallback, free — auto-selects best free model)
# Get key: https://openrouter.ai

# ── RCON Connection ───────────────────────────
# Must match Server.RconPassword in dewrito_prefs.cfg
RCON_PASSWORD=your-rcon-password

# RCON port — almost always 11776 for dedicated servers
RCON_PORT=11776

# ── Server Owner ──────────────────────────────
# Your player UUID — no 0x prefix
# Find it at http://localhost:11775 while in the server (uid field)
OWNER_UUID=your-uuid-here

# Your PrivKey from keys.cfg — copy everything BEFORE the first "+" sign
# Example: if your key is "abc123+xyz456...", put only "abc123"
OWNER_PRIVKEY=your-privkey-here


# ── Discord Bridge (optional) ─────────────────
# Leave blank if you don't want Discord integration.
# Toggle on/off in-game with: !ai discord on/off

# Webhook URL — for game chat → Discord (one direction, no bot needed)
# In Discord: Server Settings → Integrations → Webhooks → New Webhook → Copy URL
DISCORD_WEBHOOK_URL=your-webhook-url-here

# Bot Token — for Discord → game chat (both directions)
# Create a bot at https://discord.com/developers/applications
# Enable "Message Content Intent" under Bot settings
# Get key: https://discord.com/developers/applications → Bot → Token
DISCORD_BOT_TOKEN=your-discord-bot-token-here

# Channel ID — the Discord channel to bridge with the game
# Right-click the channel in Discord → Copy Channel ID (enable Developer Mode first)
DISCORD_CHANNEL_ID=your-channel-id-here
